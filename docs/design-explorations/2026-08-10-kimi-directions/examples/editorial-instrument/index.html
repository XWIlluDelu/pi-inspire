// ActivityTimeline.tsx — monotonic dwell deadlines only
const COLLAPSE_DWELL_MS = 1600;
const COMPACT_DWELL_MS = 800;

function requestCollapse(call: ToolCall, now: number) {
  const earliest = call.endedAt + COLLAPSE_DWELL_MS;
  schedule(Math.max(0, earliest - now), () => {
    if (call.phase === "settled") call.requestCollapse();
  });
}

批次侧不需要改动：requestCompaction 已经等所有卡片关闭后再起算 800 ms。剩下要做的是给 fast call 路径补一个回归测试，覆盖「完成即折叠」和「手动展开暂停批次压缩」两个用例。
